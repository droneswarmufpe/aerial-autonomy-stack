cmake_minimum_required(VERSION 3.8)
project(vio_bridge)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc)
add_executable(stereo_bridge src/stereo_bridge.cpp)
target_compile_features(stereo_bridge PUBLIC cxx_std_17)
ament_target_dependencies(stereo_bridge rclcpp sensor_msgs)
target_include_directories(stereo_bridge PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(stereo_bridge gz-transport13::gz-transport13 gz-msgs10::gz-msgs10 ${OpenCV_LIBRARIES})
install(TARGETS stereo_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
